%  Tipografía
%  TeX Gyre Pagella es un clon libre de Palatino que SÍ trae
%  versalitas y —lo importante— una fuente matemática hermana

  % IMPORTANTE: las TeX Gyre se cargan POR NOMBRE DE ARCHIVO, no por nombre
  % de familia. En macOS las fuentes del árbol de TeX no quedan registradas
  % en CoreText, de modo que \setmainfont{TeX Gyre Pagella} falla aunque el
  % .otf esté instalado, y el documento sale entero en «nullfont».
  % Por archivo, kpathsea las encuentra en cualquier plataforma.

  % --- Texto ---
  \newif\ifgyretext
  \IfFontExistsTF{texgyrepagella-regular.otf}{\gyretexttrue}{\gyretextfalse}
  \ifgyretext
    \setmainfont{texgyrepagella}[
      Extension      = .otf,
      UprightFont    = *-regular,
      ItalicFont     = *-italic,
      BoldFont       = *-bold,
      BoldItalicFont = *-bolditalic]
  \else
    \IfFontExistsTF{Palatino}{\setmainfont{Palatino}}{}
  \fi

  % --- Sans ---
  \IfFontExistsTF{texgyreheros-regular.otf}{%
    \setsansfont{texgyreheros}[
      Extension      = .otf,
      UprightFont    = *-regular,
      ItalicFont     = *-italic,
      BoldFont       = *-bold,
      BoldItalicFont = *-bolditalic,
      Scale          = MatchLowercase]%
  }{%
    \IfFontExistsTF{Helvetica Neue}{\setsansfont{Helvetica Neue}[Scale=MatchLowercase]}{}%
  }

  % --- Monoespaciada ---
  \IfFontExistsTF{texgyrecursor-regular.otf}{%
    \setmonofont{texgyrecursor}[
      Extension      = .otf,
      UprightFont    = *-regular,
      ItalicFont     = *-italic,
      BoldFont       = *-bold,
      BoldItalicFont = *-bolditalic,
      Scale          = MatchLowercase]%
  }{%
    \IfFontExistsTF{Menlo}{\setmonofont{Menlo}[Scale=MatchLowercase]}{}%
  }

  % --- Caligráfica para la dedicatoria de Agradecimientos ---
  % TeX Gyre Chorus es el clon de Zapf Chancery. La diseñó Hermann Zapf,
  % igual que Palatino, así que combina con el cuerpo del texto.
  % Si no está, cae en la itálica de la fuente principal.
  \IfFontExistsTF{texgyrechorus-mediumitalic.otf}{%
    \newfontfamily\caligrafica{texgyrechorus}[
      Extension   = .otf,
      UprightFont = *-mediumitalic,
      ItalicFont  = *-mediumitalic]%
  }{%
    \newcommand{\caligrafica}{\itshape}%
  }

  % --- Matemática: solo si existe la OTF hermana ---
  % Si falta (BasicTeX no trae tex-gyre-math), NO se carga unicode-math,
  % porque dejaría las familias matemáticas sin definir.
  % Para instalarla:  sudo tlmgr install tex-gyre tex-gyre-math

\fi

% ------------------------------------------------------------
%  Páginas de apertura de capítulo
%
%  Uso:
%  \aperturacapitulo{Título}{tamaño}{imagen}{opciones de imagen}%
%                     {epígrafe}{autor}{obra}{idioma}
% ------------------------------------------------------------
\newcommand{\aperturacapitulo}[8]{%
  \clearpage
  \refstepcounter{chapter}%
  \addcontentsline{toc}{chapter}{\protect\numberline{\thechapter}#1}%
  \chaptermark{#1}%
  \thispagestyle{empty}%
  \pagecolor{chapterbg}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north,font=\bfseries\Large\scshape,text=accent]
      (chapterlabel) at ([yshift=-34mm]current page.north)
      {Capítulo \thechapter};
    \draw[accent,line width=.5pt]
      ([xshift=-23mm]chapterlabel.west) -- (chapterlabel.west);
    \draw[accent,line width=.5pt]
      (chapterlabel.east) -- ([xshift=23mm]chapterlabel.east);
    \node[anchor=north,font=\bfseries\fontsize{#2}{#2}\selectfont,text=ink]
      at ([yshift=-54mm]current page.north) {#1};
    \node[anchor=center] at ([yshift=15mm]current page.center)
      {\includegraphics[#4]{#3}};
    \node[anchor=north,text width=.70\paperwidth,align=center,
          font=\itshape\fontsize{16}{21}\selectfont,text=ink]
      at ([yshift=-175mm]current page.north)
      {{\foreignlanguage{#8}{\hyphenchar\font=-1 ``#5''}}\\[6mm]
       {\upshape\fontsize{12}{14}\selectfont\scshape #6}\\[1mm]
       {\upshape\fontsize{12}{14}\selectfont\itshape #7}};
  \end{tikzpicture}%
  \null
  \clearpage
  \pagecolor{white}%
}
